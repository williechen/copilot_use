name: Rust AI Chat

on:
  push:
    branches: ["main"]
  pull_request:
    branches: ["main"]

env:
  CARGO_TERM_COLOR: always

jobs:
  check-chat:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Install Rust
        uses: dtolnay/rust-toolchain@stable
        with:
          targets: wasm32-unknown-unknown

      - name: Install Linux dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y \
            build-essential \
            pkg-config
            
      - name: Install cargo-leptos
        run: cargo install cargo-leptos --locked

      - name: Check server ssr
        run: cargo check --locked -p server --features ssr

      - name: Check ui ssr
        run: cargo check --locked -p ui --features ssr

      - name: Check ui
        run: cargo check --locked -p ui --features hydrate --target wasm32-unknown-unknown

  build-chat:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Install Rust
        uses: dtolnay/rust-toolchain@stable
        with:
          targets: wasm32-unknown-unknown

      - name: Install Linux dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y \
            build-essential \
            pkg-config

      - name: Install cargo-leptos
        run: cargo install cargo-leptos --locked

      - name: Build server
        run: cargo leptos build --cargo-locked --release

FILE .github/workflows/persistence.yml
name: Persistence

on:
  push:
    branches: ["main"]
  pull_request:
    branches: ["main"]

env:
  CARGO_TERM_COLOR: always

jobs:  
  test-persistence:
    runs-on: ubuntu-latest

    services:
      postgres:
        image: postgres:latest
        env:
          POSTGRES_USER: postgres
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: rust_ai_chat_test
        ports:
          - 5432:5432
        options: >-
          --health-cmd "pg_isready -U postgres -d rust_ai_chat_test"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    env:
      ANALYTICS_DATABASE_URL: postgres://postgres:postgres@localhost:5432/rust_ai_chat_test
      ANALYTICS_CHAT_TOKEN: local-ci-chat-token
      ANALYTICS_PET_TOKEN: local-ci-pet-token
      CHAT_DATABASE_URL: "sqlite::memory:"
      PET_DATABASE_URL: "sqlite::memory:"

    steps:
      - uses: actions/checkout@v4

      - name: Install Rust
        uses: dtolnay/rust-toolchain@stable

      - name: Install sqlx-cli
        run: cargo install sqlx-cli --version "^0.9" --no-default-features --features postgres,sqlite --locked

      - name: Run database migrations
        run: sqlx migrate run --database-url $CHAT_DATABASE_URL

      - name: Test persistence
        run: DATABASE_URL="$CHAT_DATABASE_URL" cargo test --locked -p persistence

      - name: Test stats server with PostgreSQL
        run: DATABASE_URL="$ANALYTICS_DATABASE_URL" cargo test --locked -p stats-server --features postgres-integration --test postgres_event_store
