```rs
struct ExistingRoom;
#[async_trait]
impl RoomRepository for ExistingRoom {
    async fn find_by_id(&self, id: &str) -> Result<Option<Room>, RepositoryError> {
        Ok(Some(Room { id: id.into(), category_room_id: "c1".into(), name: "room".into() }))
    }
    async fn create(&self, _: &Room) -> Result<(), RepositoryError> { Ok(()) }
}
struct MissingRoom;
#[async_trait]
impl RoomRepository for MissingRoom {
    async fn find_by_id(&self, _: &str) -> Result<Option<Room>, RepositoryError> { Ok(None) }
    async fn create(&self, _: &Room) -> Result<(), RepositoryError> { Ok(()) }
}
#[tokio::test]
async fn missing_room_blocks_message_and_publish() {
    let mut calls = 0;
    let msg = ChatMessage { id: "m".into(), room_id: "absent".into(), author: MessageAuthor::AnonymousUser, content: "hi".into() };
    let result = persist_in_existing_room(&MissingRoom, &SuccessfulRepository, msg, |_| calls += 1).await;
    assert!(matches!(result, Err(RepositoryError::RoomAccessError(_))));
    assert_eq!(calls, 0);
}
#[tokio::test]
async fn existing_room_allows_persist_and_publish() {
    let mut calls = 0;
    let msg = ChatMessage { id: "m".into(), room_id: "r1".into(), author: MessageAuthor::AnonymousUser, content: "hi".into() };
    persist_in_existing_room(&ExistingRoom, &SuccessfulRepository, msg, |_| calls += 1).await.unwrap();
    assert_eq!(calls, 1);
}
```

Green：在既有 <code>persist_then_publish</code> 下方加入：

```rs
pub async fn persist_in_existing_room(
    rooms: &dyn RoomRepository,
    messages: &dyn MessageRepository,
    message: ChatMessage,
    publish: impl FnOnce(ChatMessage),
) -> Result<(), RepositoryError> {
    if rooms.find_by_id(&message.room_id).await?.is_none() {
        return Err(RepositoryError::RoomAccessError("房間不存在".into()));
    }
    persist_then_publish(messages, message, publish).await
}
```
