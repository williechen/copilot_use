```rs
#![cfg(feature = "postgres-integration")]
use sqlx::PgPool;
use stats_server::{EventInput, pg_store::{persist_event, PersistOutcome, PersistError}};
use uuid::Uuid;

#[sqlx::test(migrations = "./migrations")]
async fn postgres_event_idempotency(pool: PgPool) {
    let id = Uuid::new_v4();
    let event = EventInput {
        schema_version: 1,
        event_id: id,
        source_app: "web_chat".into(),
        event_type: "chat.message.persisted".into(),
    };
    assert_eq!(persist_event(&pool, &event).await.unwrap(), PersistOutcome::Stored);
    assert_eq!(persist_event(&pool, &event).await.unwrap(), PersistOutcome::AlreadyExists);
    let conflict = EventInput {
        schema_version: 1,
        event_id: id,
        source_app: "web_chat".into(),
        event_type: "chat.message.changed".into(),
    };
    assert_eq!(persist_event(&pool, &conflict).await.unwrap(), PersistOutcome::IdentityConflict);
    let pet = EventInput {
        schema_version: 1,
        event_id: id,
        source_app: "desktop_pet".into(),
        event_type: "pet.cleanup.succeeded".into(),
    };
    assert_eq!(persist_event(&pool, &pet).await.unwrap(), PersistOutcome::Stored);
    let invalid = EventInput {
        schema_version: 2,
        event_id: Uuid::new_v4(),
        source_app: "web_chat".into(),
        event_type: "chat.message.persisted".into(),
    };
    assert!(matches!(persist_event(&pool, &invalid).await, Err(PersistError::Invalid)));
    let count: (i64,) = sqlx::query_as("SELECT COUNT(*) FROM product_events")
        .fetch_one(&pool).await.unwrap();
    assert_eq!(count.0, 2);
}
```

此補加是原 Day 11「PostgreSQL 事件儲存與冪等」的必要介面說明，<b>不是宣稱 HTTP 攝取已實作或驗收</b>。複核 rust\_ai\_chat/main <code>5a0a8cee0d1fb4d0cd248869d7b24ebb1d653399</code>：

- 缺少：<code>build_app</code> 未持有／注入 <code>PgPool</code>；<code>ingest</code> 未取得資料庫狀態，也未呼叫 <code>pg_store::persist_event</code>，有效事件仍固定回 <code>501 NOT_IMPLEMENTED</code>；沒有來源身分認證，不能直接將有效 JSON 視為可信來源。此處的 <code>source_app</code> 僅是客戶端可填的欄位，不等於認證。
- 預期完整呼叫路徑：經來源身分驗證的請求 → <code>POST /api/v1/events</code> → <code>ingest(State<IngestState>, Json<EventInput>)</code> → <code>valid_event</code> → 驗證身分與 <code>source_app</code> 一致 → <code>persist_event(&state.pool, &event)</code> → 將 <code>Stored</code>、<code>AlreadyExists</code>、<code>IdentityConflict</code> 與資料庫錯誤分別轉換為明確 HTTP 回應。HTTP ACK 應在資料庫操作成功後才回傳。
- 後續最小功能切片：先設計可信來源驗證及 <code>PgPool</code> 生命週期，再實作 HTTP Adapter、回應契約與隔離 PostgreSQL 整合測試；最後才接 gRPC／Protobuf 上報及 JSON／SSE Dashboard。禁止在缺少來源認證時僅把 <code>501</code> 改為 <code>200</code>。
