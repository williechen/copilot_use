### Green A：共用 token/Cookie 規則

<code>crates/shared/Cargo.toml</code> 的 \[dependencies\] 增加 <code>sha2.workspace = true</code>，<code>crates/shared/src/lib.rs</code> 增加 <code>pub mod auth;</code>。新增 <code>crates/shared/src/auth.rs</code> 完整內容：

```rs
use sha2::{Digest, Sha256};

pub fn cookie_token(header: &str, name: &str) -> Option<String> {
    let value = header.split(';').filter_map(|part| part.trim().split_once('='))
        .find(|(key, _)| *key == name)?.1;
    (value.len() == 64 && value.bytes().all(|b| b.is_ascii_hexdigit()))
        .then(|| value.to_owned())
}

pub fn token_digest(token: &str) -> String {
    Sha256::digest(token.as_bytes()).iter()
        .map(|b| format!("{b:02x}")).collect()
}

#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn cookie_name_is_exact_and_token_is_validated() {
        let token = "a".repeat(64);
        assert_eq!(cookie_token(&format!("xchat_session={token}; chat_session={token}"), "chat_session"), Some(token.clone()));
        assert_eq!(cookie_token("chat_session=bad", "chat_session"), None);
        assert_eq!(cookie_token("analytics_session=abc", "chat_session"), None);
        assert_eq!(token_digest(&token).len(), 64);
    }
}
```

此共用模組不認識 Chat 或 Analytics DB；Cookie 名稱由呼叫端傳入，兩系統仍使用不同 Cookie 名稱與不同資料庫。<b>不要把 token hash 或 Cookie 解析再次貼進各系統新 Handler。</b>

### Green B：登出路由（含 Origin 與表單 CSRF）

新增 <code>apps/server/src/logout.rs</code>：

```rs
use axum::{
    Form, Router,
    extract::State,
    http::{HeaderMap, StatusCode, header},
    response::{IntoResponse, Response},
    routing::post,
};
use serde::Deserialize;
use shared::auth::{cookie_token, token_digest};
use sqlx::SqlitePool;

#[derive(Deserialize)]
struct LogoutForm { csrf: String }

pub fn routes(pool: SqlitePool) -> Router {
    Router::new().route("/logout", post(logout)).with_state(pool)
}

async fn logout(
    State(pool): State<SqlitePool>,
    headers: HeaderMap,
    Form(form): Form<LogoutForm>,
) -> Result<Response, StatusCode> {
    if headers.get(header::ORIGIN).and_then(|v| v.to_str().ok())
        != Some("http://127.0.0.1:3000") {
        return Err(StatusCode::FORBIDDEN);
    }
    let token = headers.get(header::COOKIE).and_then(|v| v.to_str().ok())
        .and_then(|h| cookie_token(h, "chat_session"))
        .ok_or(StatusCode::UNAUTHORIZED)?;
    if form.csrf != token_digest(&format!("logout:{token}")) {
        return Err(StatusCode::FORBIDDEN);
    }
    let result = sqlx::query("DELETE FROM mbr_session WHERE token_hash=?1")
        .bind(token_digest(&token)).execute(&pool).await
        .map_err(|_| StatusCode::SERVICE_UNAVAILABLE)?;
    if result.rows_affected() != 1 {
        return Err(StatusCode::UNAUTHORIZED);
    }
    Ok((
        StatusCode::SEE_OTHER,
        [(header::LOCATION, "/"),
         (header::SET_COOKIE, "chat_session=; HttpOnly; SameSite=Lax; Path=/; Max-Age=0"),
         (header::CACHE_CONTROL, "no-store")],
    ).into_response())
}
```

<code>apps/server/src/lib.rs</code> 增加 <code>pub mod logout;</code>，在 <code>application()</code> 內既有 <code>let signin_routes = ...;</code> 之後新增：

```rs
let logout_routes = logout::routes(db.clone());
```

於 Router 原有 <code>.merge(signin_routes)</code> 後再接 <code>.merge(logout_routes)</code>。CSRF 值不存新資料表，由既有 Session token 導出；POST 驗證 Origin，拒絕跨站提交。正式部署如不在 loopback 3000，必須將既有 login/signup 及此路由改為一致的可信 Origin 政策。Cookie 若改在 HTTPS，須統一加入 Secure 屬性。

use server::{signin,logout};

fn hash(t: &str) -> String {
    Sha256::digest(t.as_bytes()).iter().map(|b| format!("{b:02x}")).collect()
}

#[tokio::test]
async fn day26_logout_revokes_only_one_session() {
    let db = sqlx::sqlite::SqlitePoolOptions::new().max_connections(1)
        .connect("sqlite::memory:").await.unwrap();
    sqlx::raw_sql(include_str!("../../../migrations/20261018073931_create_auth_user.sql"))

    sqlx::raw_sql(include_str!("../../../migrations/20261020152737_mbr_sessions.sql"))

    sqlx::query("INSERT INTO mbr_user (id,user_acnt,user_pd,user_email,user_name,user_status,user_sexy,user_birthday,is_init_pd,created_at,updated_at) VALUES('m1','user','hash','u@test.example','User','active',NULL,NULL,FALSE,CURRENT_TIMESTAMP,CURRENT_TIMESTAMP)")
        .execute(&db).await.unwrap();
    let a = "a".repeat(64); let b = "b".repeat(64);
    for (id,raw) in [("s1",&a),("s2",&b)] {
        sqlx::query("INSERT INTO mbr_session(id,mbr_user_id,token_hash,expires_at) VALUES(?1,'m1',?2,unixepoch()+3600)")
            .bind(id).bind(hash(raw)).execute(&db).await.unwrap();
    }
    let app = axum::Router::new().merge(signin::routes(db.clone()))
        .merge(logout::routes(db.clone()));
    let req = |raw: &str| Request::get("/account")
        .header(header::COOKIE,format!("chat_session={raw}"))
        .body(Body::empty()).unwrap();
    assert_eq!(app.clone().oneshot(req(&a)).await.unwrap().status(), StatusCode::OK);
    let csrf = hash(&format!("logout:{a}"));
    let bad = app.clone().oneshot(Request::post("/logout")
        .header(header::ORIGIN,"http://127.0.0.1:3000")
        .header(header::COOKIE,format!("chat_session={a}"))
        .header(header::CONTENT_TYPE,"application/x-www-form-urlencoded")
        .body(Body::from("csrf=wrong")).unwrap()).await.unwrap();
    assert_eq!(bad.status(),StatusCode::FORBIDDEN);
    let ok = app.clone().oneshot(Request::post("/logout")
        .header(header::ORIGIN,"http://127.0.0.1:3000")
        .header(header::COOKIE,format!("chat_session={a}"))
        .header(header::CONTENT_TYPE,"application/x-www-form-urlencoded")
        .body(Body::from(format!("csrf={csrf}"))).unwrap()).await.unwrap();
    assert_eq!(ok.status(),StatusCode::SEE_OTHER);
    assert!(ok.headers()[header::SET_COOKIE].to_str().unwrap().contains("Max-Age=0"));
    assert_eq!(app.clone().oneshot(req(&a)).await.unwrap().status(),StatusCode::UNAUTHORIZED);
    assert_eq!(app.oneshot(req(&b)).await.unwrap().status(),StatusCode::OK);
}
