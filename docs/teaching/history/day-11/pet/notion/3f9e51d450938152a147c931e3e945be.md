// 替換原 scan_file_preview 函式，並新增同檔命令；
// 既有 file_organizer::scan_preview 保留，避免 dead_code。
fn authorized_root() -> Result<file_organizer::AuthorizedRoot, String> {
    let path = std::env::var("PET_AUTHORIZED_ROOT")
        .map_err(|_| "尚未設定授權掃描目錄".to_string())?;
    file_organizer::AuthorizedRoot::new(path)
        .map_err(|e| format!("無法使用授權掃描目錄：{e}"))
}

#[tauri::command]
fn scan_file_preview() -> Result<file_organizer::ScanPreview, String> {
    let root = authorized_root()?;
    file_organizer::scan_preview(&root)
        .map_err(|e| format!("掃描失敗：{e}"))
}

#[tauri::command]
fn scan_duplicate_preview() -> Result<file_organizer::DuplicatePreview, String> {
    let root = authorized_root()?;
    let scan = file_organizer::scan(&root)
        .map_err(|e| format!("掃描失敗：{e}"))?;
    // 僅回傳 SHA256 分組與警告；不提供刪除或 Trash 能力。
    Ok(file_organizer::duplicate_preview(&scan))
}

<b>最小 Green（WASM bridge）：</b> 沿用既有 wasm-bindgen、JsFuture、serde-wasm-bindgen，不讓 UI 持有實際檔案路徑。

```rs
// apps/desktop_pet/frontend/src/bridge.rs
// 加在既有 #[wasm_bindgen] extern "C" { ... } 區塊內：
#[wasm_bindgen(js_namespace = window, js_name = scanDuplicatePreview)]
fn scan_duplicate_preview_js() -> Promise;

// 在 extern 區塊外加入：只向 UI 傳群組與警告數量。
#[derive(serde::Deserialize)]
struct DuplicatePreviewCounts {
    groups: Vec<serde::de::IgnoredAny>,
    warnings: Vec<serde::de::IgnoredAny>,
}

pub async fn scan_duplicate_preview() -> Result<(usize, usize), String> {
    let value = JsFuture::from(scan_duplicate_preview_js())
        .await
        .map_err(js_error)?;
    let result: DuplicatePreviewCounts =
        serde_wasm_bindgen::from_value(value).map_err(|e| e.to_string())?;
    Ok((result.groups.len(), result.warnings.len()))
}
```
