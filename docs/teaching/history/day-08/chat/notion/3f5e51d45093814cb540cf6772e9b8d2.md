/// 僅包含產品分析需要的最小欄位；不攜帶訊息正文、房間或使用者 ID。
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PersistedMessageEvent {
    pub schema_version: u16,
    pub event_id: Uuid,
    pub source_app: &'static str,
    pub event_type: &'static str,
}

/// 使用持久化訊息的 ID 當冪等鍵，重試時不可每次產生新 UUID。
/// 不合法的訊息 ID 代表資料契約錯誤，必須回報而非偷偷創建新事件 ID。
pub fn persisted_message_event(message: &ChatMessage) -> Result<PersistedMessageEvent, uuid::Error> {
    Ok(PersistedMessageEvent {
        schema_version: 1,
        event_id: Uuid::parse_str(&message.id)?,
        source_app: "web_chat",
        event_type: "message.persisted",
    })
}

#[derive(Debug, Error)]
pub enum RoomRepositoryError {

    Repository(String),

#[async_trait]
pub trait RoomRepository: Send + Sync {
    async fn find_by_id(&self, room_id: &str) -> Result<Option<Room>, RoomRepositoryError>;

    async fn create(&self, room: &Room) -> Result<(), RoomRepositoryError>;
}

#[derive(Debug, thiserror::Error)]
pub enum MessageRepositoryError {
    #[error("message repository failure: {0}")]
    Repository(String),
}

#[async_trait]
pub trait MessageRepository: Send + Sync {
    async fn create(&self, message: &ChatMessage) -> Result<(), MessageRepositoryError>;
}

/// 先持久化再通知；資料庫失敗時 closure 絕不執行。
pub async fn persist_then_publish(
    repository: &dyn MessageRepository,
    message: ChatMessage,
    publish: impl FnOnce(ChatMessage),
) -> Result<(), MessageRepositoryError> {
    repository.create(&message).await?;
    publish(message);
    Ok(())
}

use chat_application::{
    MessageRepository, MessageRepositoryError, persist_then_publish,
    product_event::persisted_message_event,
};
use chat_domain::{ChatMessage, MessageAuthor};
use std::sync::Mutex;
use uuid::Uuid;

struct FakeRepository { fail: bool, saved: Mutex<Vec<ChatMessage>> }

#[async_trait]
impl MessageRepository for FakeRepository {
    async fn create(&self, message: &ChatMessage) -> Result<(), MessageRepositoryError> {
        if self.fail {
            return Err(MessageRepositoryError::Repository("模擬儲存失敗".into()));
        }
        self.saved.lock().unwrap().push(message.clone());
        Ok(())
    }
}

fn message() -> ChatMessage {
    ChatMessage {
        id: Uuid::from_u128(1).to_string(),
        room_id: "private-room".into(),
        author: MessageAuthor::AnonymousUser,
        content: "不可外送的私人訊息".into(),
    }

#[tokio::test]
async fn success_generates_stable_minimal_event_only_after_persist() {
    let repo = FakeRepository { fail: false, saved: Mutex::new(vec![]) };
    let events = Mutex::new(Vec::new());
    let input = message();
    persist_then_publish(&repo, input.clone(), |saved| {
        // 只有成功儲存之後，才允許生成產品事件。
        assert_eq!(repo.saved.lock().unwrap().len(), 1);
        events.lock().unwrap().push(persisted_message_event(&saved).unwrap());
    }).await.unwrap();
    let events = events.lock().unwrap();
    assert_eq!(events.len(), 1);
    assert_eq!(events[0].event_id.to_string(), input.id);
    assert_eq!(events[0].schema_version, 1);
    assert_eq!(events[0].source_app, "web_chat");
    assert_eq!(events[0].event_type, "message.persisted");
}

#[tokio::test]
async fn database_failure_never_generates_event() {
    let repo = FakeRepository { fail: true, saved: Mutex::new(vec![]) };
    let events = Mutex::new(0);
    let result = persist_then_publish(&repo, message(), |saved| {
        if persisted_message_event(&saved).is_ok() {
            *events.lock().unwrap() += 1;
        }
    }).await;
    assert!(result.is_err());
    assert_eq!(*events.lock().unwrap(), 0);

use crate::{state::AppState, ws_policy::{room_matches_connection, valid_message_content}};
use axum::{
    extract::{
        Path, State,
        ws::{Message, WebSocket, WebSocketUpgrade},
    },
    response::Response,
};
use shared::{ClientEvent, ServerEvent};
use tokio::sync::broadcast;

// 訊息內容不可僅有空白；保留原始內容，不自動裁切。
fn valid_message_content(content: &str) -> bool {
    !content.trim().is_empty()
}

// 防止跨房間發送；空白識別碼不可視為合法房間。
fn room_matches_connection(connection: &str, event: &str) -> bool {
    !connection.trim().is_empty() && !event.trim().is_empty() && connection == event
}

async fn send_event(
    socket: &mut WebSocket,
    event: ServerEvent,
) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
    let json = serde_json::to_string(&event)?;
    socket.send(Message::Text(json.into())).await?;
    Ok(())
}

pub async fn ws_handler(
    Path(room_id): Path<String>,
    State(state): State<AppState>,
    ws: WebSocketUpgrade,
) -> Response {
    match state.room_repository.find_by_id(&room_id).await {
        Ok(Some(_room)) => ws.on_upgrade(move |socket| handle_socket(socket, state, room_id)),
        Ok(None) => Response::builder()
            .status(404)
            .body("Room not found".into())
            .unwrap(),
        Err(error) => {
            eprintln!("websocket room lookup error: {:?}", error);
            Response::builder()
                .status(500)
                .body("Internal server error".into())
                .unwrap()
        }
    }
}

async fn handle_socket(mut socket: WebSocket, state: AppState, room_id: String) {
    let (room_sender, mut room_events) = state.room_hub.subscribe(room_id.clone());

    loop {
        tokio::select! {
            incoming = socket.recv() => {
                let Some(result) = incoming else {
                    break;
                };
                let message = match result {
                    Ok(message) => message,
                    Err(error) => {
                        eprintln!("websocket receive error: {error}");
                        break;
                    }
                };

                if let Message::Text(text) = message
                    && let Err(error) = handle_client_event(&mut socket, &state, text.as_str(), room_id.clone()).await {
                        eprintln!("client message error: {error}");
                        break;
                }
            }
            event = room_events.recv() => {
                match event {
                    Ok(event) => {
                        if let Err(error) = send_event(&mut socket, event).await {
                            eprintln!("websocket send error: {error}");
                            break;
                        }
                    }
                    Err(broadcast::error::RecvError::Lagged(skipped)) => {
                        eprintln!("websocket receiver lagged skipped: {skipped} events");
                    }
                    Err(broadcast::error::RecvError::Closed) => {
                        eprintln!("websocket receiver closed");
                        break;
                    }
                }
            }
        }
    }

    drop(room_events);

    state.room_hub.cleanup(room_id.clone(), &room_sender);
}

async fn handle_client_event(
    socket: &mut WebSocket,
    state: &AppState,
    text: &str,
    event_room_id: String,
) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
    let event: ClientEvent = serde_json::from_str(text)?;

    match event {
        ClientEvent::Ping => {
            send_event(socket, ServerEvent::Pong).await?;
        }
        ClientEvent::SendMessage { room_id, content } => {
            if !room_matches_connection(&event_room_id, &room_id) {
                send_event(
                    socket,
                    ServerEvent::Error {
                        code: "room_mismatch".to_string(),
                        message: "message room does not match connection room".to_string(),
                    },
                )
                .await?;
                return Ok(());
            }

            if !valid_message_content(&content) {
                send_event(
                    socket,
                    ServerEvent::Error {
                        code: "invalid_message_content".to_owned(),
                        message: "訊息不可為空白".to_owned(),
                    },
                )
                .await?;
                return Ok(());
            }

            let message = state.chat_service.send_message(room_id.clone(), content);
            if let Err(error) = chat_application::persist_then_publish(
                state.message_repository.as_ref(),
                message,
                |saved| {
                    state
                        .room_hub
                        .publish(room_id.clone(), ServerEvent::MessageCreated(saved))
                },
            )
            .await
            {
                eprintln!("message persistence failed: {error}");
                send_event(
                    socket,
                    ServerEvent::Error {
                        code: "message_persistence_failed".to_owned(),
                        message: "訊息儲存失敗，請稍後重試".to_owned(),
                    },
                )
                .await?;
                return Ok(());
            }
        }
        ClientEvent::Typing { .. } => {
            // 後面再做 presence / typing。
        }
    }

    Ok(())
}
